\section{Background: TICL over Imp}
% ============================================================================

\begin{frame}{TICL Structural Lemmas for Imp}
  \begin{block}{Core idea (from the TICL paper)}
    For a small imperative language \stimp with mutable state,
    the entailment $\sentailsL{S}{t}{m}{\varphi}$ is proved
    structurally via inference rules on the syntax of $t$.
  \end{block}

  \vspace{0.3cm}
  Key rules from \texttt{imp-lemmas.tex}:
  \begin{itemize}
    \item \textbf{Seq}$_S$AU$_L$: sequential composition decomposes into bind
    \item \textbf{If}$_S$AU$_L$: conditional branches both satisfy the obligation
    \item \textbf{While}$_S$AG: infinite loop with coinductive invariant $\R$
    \item \textbf{While}$_S$AU$_L$: terminating loop with ranking function $f$
  \end{itemize}

  \vspace{0.3cm}
  \alert{Goal:} Lift these to Rust syntax $\to$ define $\mathtt{wp}$ for Vera.
\end{frame}

\begin{frame}{Notation}
  \begin{align*}
    \sigma &\in \text{State (mutable references, heap)} \\
    \ventailsL{t}{\sigma}{\varphi} &\triangleq \text{``Rust term } t \text{ from state } \sigma \text{ satisfies } \varphi\text{''} \\
    \wpx{t}{\varphi} &\triangleq \text{weakest precondition on } \sigma \text{ s.t.\ } \ventailsL{t}{\sigma}{\varphi}
  \end{align*}

  \begin{block}{Temporal formulas}
    \begin{itemize}
      \item $\varphi ::= \now{P} \mid \AG{\varphi} \mid \AU{\varphi}{\varphi'} \mid \AN{\varphi}{\varphi'} \mid \varphi \land \varphi' \mid \varphi \lor \varphi' \mid \varphi \implies \varphi'$
      \item Sugar: $\AF{\varphi'} = \AU{\top}{\varphi'}$, \quad $\AX{\varphi'} = \AN{\top}{\varphi'}$
      \item $\now{P}$ where $P : \sigma \to \mathbb{P}$ --- state predicate on current state
      \item $\done{Q}$ where $Q : \tau \to \sigma \to \mathbb{P}$ --- predicate on return value and final state
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Comparison with Hoare Logic}
  \begin{block}{Total correctness Hoare triple}
    $$\{P\}~c~\{Q\} \quad\iff\quad \forall~\sigma.~ P~\sigma \to \ventailsL{c}{\sigma}{\AF{\done{Q}}}$$
  \end{block}

  Hoare logic is a special case of TICL:
  \begin{itemize}
    \item $\AF{\done{Q}} = \AU{\top}{\done{Q}}$: trivial path, just reach $Q$
    \item $P$ is the precondition (state predicate on $\sigma$)
    \item $Q$ is the postcondition (predicate on return value and final state)
  \end{itemize}

  \vspace{0.3cm}
  \begin{block}{What TICL adds beyond Hoare logic}
    \begin{tabular}{l l}
      $\AG{\varphi}$ & Safety invariant (infinite computation) \\
      $\AU{\varphi}{\done{Q}}$ & Path property $\varphi$ until $Q$ \\
      $\AG{\AF{\now{P}}}$ & Fairness / recurrence \\
      $\AN{\varphi}{\done{Q}}$ & One-step (next) \\
    \end{tabular}
  \end{block}
\end{frame}

\section{Inference Rules (Single-Process)}
% ============================================================================

\begin{frame}{Assignment}
  \begin{block}{Rust: \quad $\rmut{x}{e}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=AssignAU$_L$]
    {\ventailsR{e}{\sigma}{\AU{\varphi}{\done{\R}}} \\\\
     \forall~v, \sigma',~\R~v~\sigma' \to \ventailsL{\ISkip}{\sigma'[x \mapsto v]}{\AU{\varphi}{\varphi'}}}
    {\ventailsL{\rmut{x}{e}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
  Evaluating $e$ may take steps; $\varphi$ must hold along the path.
\end{frame}

\begin{frame}{Sequential Composition (let-binding)}
  \begin{block}{Rust: \quad $\rlet{x}{t}{u}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=LetAU$_L$]
    {\ventailsR{t}{\sigma}{\AU{\varphi}{\done{\R}}} \\\\
     \forall~x,\sigma',~\R~x~\sigma' \to \ventailsL{u[x]}{\sigma'}{\AU{\varphi}{\varphi'}}}
    {\ventailsL{\rlet{x}{t}{u}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
  Direct lift of \textbf{BindAU$_L$} / \textbf{Seq}$_S$AU$_L$ from TICL.
\end{frame}

\begin{frame}{Conditional}
  \begin{block}{Rust: \quad $\rif{c}{t}{u}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=IfAU$_L$]
    {\ventailsR{c}{\sigma}{\AU{\varphi}{\doneq{(b,\sigma')}{}}} \\\\
     {
       \begin{cases}
         \ventailsL{t}{\sigma'}{\AU{\varphi}{\varphi'}}, &\text{if } b \\
         \ventailsL{u}{\sigma'}{\AU{\varphi}{\varphi'}}, &\text{otherwise}
       \end{cases}
     }
    }
    {\ventailsL{\rif{c}{t}{u}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
  Evaluating $c$ may take multiple steps; $\varphi$ must hold along the path.
\end{frame}

\begin{frame}{While Loop (AU -- terminating)}
  \begin{block}{Rust: \quad $\rwhile{c}{t}$ \quad with \texttt{invariant} $\R$, \texttt{decreases} $f$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=WhileAU$_L$]
    {\R~\sigma \\\\
     {\begin{aligned}
      \forall~\sigma,~&\R~\sigma \to \ventailsR{c}{\sigma}{\AU{\varphi}{\doneq{(b,\sigma')}{}}} \\
      & \land \begin{cases}
        \ventailsL{t}{\sigma'}{\AU{\varphi}{\varphi'}} ~\lor \\
        \quad \ventailsR{t}{\sigma'}{\AU{\varphi}{\done{(\lambda~\sigma'' .~
            \R~\sigma'' \land f~\sigma'' < f~\sigma)}}}, &\text{if } b \\
        \varphi'(\sigma'), &\text{otherwise}
      \end{cases}
     \end{aligned}}}
    {\ventailsL{\rwhile{c}{t}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
\end{frame}

\begin{frame}{While Loop (AG -- infinite)}
  \begin{block}{Rust: \quad $\rwhile{c}{t}$ \quad with \texttt{invariant} $\R$ (no \texttt{decreases})}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=WhileAG]
    {\R~\sigma \\\\
     {\begin{aligned}
      \forall~\sigma,~&\R~\sigma \to \\
      & \ventailsL{\rwhile{c}{t}}{\sigma}{\varphi} ~\land \\
      & \ventailsR{c}{\sigma}{\AU{\varphi}{\doneq{(\tm{true},\sigma)}{}}} ~\land \\
      & \ventailsR{t}{\sigma}{\varphi~\AU{}~\done{(\lambda~\sigma' \Rightarrow \R~\sigma')}}
     \end{aligned}}}
    {\ventailsL{\rwhile{c}{t}}{\sigma}{\AG{\varphi}}}
  \end{mathpar}
  }
  Coinductive hypothesis $\ventailsL{\rwhile{c}{t}}{\sigma}{\varphi}$; condition must always be true.
\end{frame}

\begin{frame}{Function Call}
  \begin{block}{Rust: \quad $\rcall{f}{e_1, \ldots, e_n}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=CallAU$_L$]
    {\ventailsR{e_1}{\sigma_0}{\AU{\varphi}{\done{\R_1}}} \\\\
     \forall~v_1,\sigma_1.~ \R_1~v_1~\sigma_1 \to \ventailsR{e_2}{\sigma_1}{\AU{\varphi}{\done{\R_2}}} \\\\
     \cdots \\\\
     \forall~v_n,\sigma_n.~ \R_n~v_n~\sigma_n \to \ventailsR{\rcall{f}{v_1, \ldots, v_n}}{\sigma_n}{\AU{\varphi}{\done{\R}}}}
    {\ventailsR{\rcall{f}{e_1, \ldots, e_n}}{\sigma_0}{\AU{\varphi}{\done{\R}}}}
  \end{mathpar}
  }
  Arguments evaluated left-to-right; $\R_i$ chains $\sigma_i$ into $e_{i+1}$.
  Call produces result via $\done{\R}$; caller binds it with let.
\end{frame}

\section{Single-Process Weakest Preconditions}
% ============================================================================

\begin{frame}{wp: Assignment}
    \begin{align*}
    \wpx{\rmut{x}{e}}{\AU{\varphi}{\varphi'}} ~=~ \lambda~\sigma.~
    &~ \wpx{e}{\AU{\varphi}{\done{\R}}}~\sigma \\
    \land~& ~\forall~v, \sigma'.~ \R~v~\sigma' \to \wpx{\ISkip}{\AU{\varphi}{\varphi'}}~(\sigma'[x \mapsto v])
    \end{align*}
\end{frame}

\begin{frame}{wp: Let-binding (Sequential Composition)}
    \begin{align*}
    \wpx{\rlet{x}{t}{u}}{\AU{\varphi}{\varphi'}} ~=~ \lambda~\sigma.~
    &~ \wpx{t}{\AU{\varphi}{\done{\R}}}~\sigma \\
    \land~& ~\forall x, \sigma'.~ \R~x~\sigma' \to \wpx{u[x]}{\AU{\varphi}{\varphi'}}~\sigma'
    \end{align*}
\end{frame}

\begin{frame}{wp: Conditional}
    \begin{align*}
    &\wpx{\rif{c}{t}{u}}{\AU{\varphi}{\varphi'}} ~=~ \lambda~\sigma. \\
    & \quad \wpx{c}{\AU{\varphi}{\done{\R}}}~\sigma ~\land~ \forall~b,\sigma'.~ \R~b~\sigma' \to \\
    & \quad \begin{cases}
      \wpx{t}{\AU{\varphi}{\varphi'}}~\sigma', &\text{if } b \\
      \wpx{u}{\AU{\varphi}{\varphi'}}~\sigma', &\text{otherwise}
    \end{cases}
    \end{align*}
\end{frame}

\begin{frame}{wp: While Loop (AU)}
    \begin{align*}
    &\wpx{\rwhile{c}{t}}{\AU{\varphi}{\varphi'}} ~=~ \lambda~\sigma. \\
    & \quad \R~\sigma ~\land~ \forall~\sigma,~\R~\sigma \to \\
    & \quad \wpx{c}{\AU{\varphi}{\done{\R'}}}~\sigma ~\land~ \forall~b,\sigma'.~ \R'~b~\sigma' \to \\
    & \quad \begin{cases}
      \wpx{t}{\AU{\varphi}{\varphi'}}~\sigma' ~\lor \\
      \quad \wpx{t}{\AU{\varphi}{\done{(\lambda~\_~\sigma''.~
          \R~\sigma'' \land f~\sigma'' < f~\sigma)}}}~\sigma', &\text{if } b \\
      \varphi'(\sigma'), &\text{o.w.}
    \end{cases}
    \end{align*}
\end{frame}

\begin{frame}{wp: While Loop (AG)}
    \begin{align*}
    &\wpx{\rwhile{c}{t}}{\AG{\varphi}} ~=~ \lambda~\sigma. \\
    & \quad \R~\sigma ~\land~ \forall~\sigma,~\R~\sigma \to \\
    & \quad \wpx{\rwhile{c}{t}}{\varphi}~\sigma ~\land \\
    & \quad \wpx{c}{\AU{\varphi}{\doneq{(\tm{true},\sigma)}{}}}~\sigma ~\land \\
    & \quad \wpx{t}{\varphi~\AU{}~\done{(\lambda~\sigma'.~ \R~\sigma')}}~\sigma
    \end{align*}
    Coinductive: $\wpx{\rwhile{c}{t}}{\varphi}~\sigma$ is the coinductive hypothesis.
\end{frame}

\begin{frame}{wp: Function Call -- $f$ ensures $\AF{\done{\R_f}}$}
  \begin{block}{$f$ has terminating postcondition}
    \begin{align*}
    &\wpx{\rcall{f}{e_1, \ldots, e_n}}{\AU{\varphi}{\done{\R}}} ~=~ \lambda~\sigma_0. \\
    &\quad \wpx{e_1}{\AU{\varphi}{\done{\R_1}}}~\sigma_0 ~\land~ \forall~v_1,\sigma_1.~ \R_1~v_1~\sigma_1 \to \\
    &\quad \cdots \\
    &\quad \forall~v_n,\sigma_n.~ \R_n~v_n~\sigma_n \to \wpx{\rcall{f}{v_1,\ldots,v_n}}{\AU{\varphi}{\done{\R}}}~\sigma_n
    \end{align*}
  \end{block}

  \begin{block}{$f$ has temporal ensures $\psi$: transparent checking}
    Additionally assert $\psi \implies \varphi$ where $\varphi$ is the caller's obligation.
  \end{block}

  \begin{block}{$f$ ensures $\AG{\psi}$: cannot prove $\AF{}$ for it}
    If $f$ has $\AG{\psi}$ (diverges), then $\wpx{\rcall{f}{\vec{e}}}{\AF{\varphi'}}$ is \textbf{unprovable} --
    a diverging call can never reach the $\AF{}$ goal.
  \end{block}
\end{frame}

\section{Vera Implementation}
% ============================================================================

\begin{frame}[fragile]{Vera Syntax}
  Vera extends Verus with temporal postconditions and loop annotations:

  \vspace{0.2cm}
  \begin{tabular}{l l l}
    \textbf{Annotation} & \textbf{TICL} & \textbf{Proof obligation} \\
    \hline
    \verb|ensures af(done(Q))| & $\AF{\done{Q}}$ & Loop + \verb|decreases| \\
    \verb|ensures ag(P)| & $\AG{\varphi}$ & Infinite loop + \verb|invariant| \\
    \verb|ensures au(P, done(Q))| & $\AU{P}{\done{Q}}$ & Loop + \verb|invariant| + \verb|decreases| \\
    \verb|ensures ag(af(now(Q)))| & $\AG{\AF{\now{Q}}}$ & Ghost accumulator + \verb|decreases| \\
    \hline
    \verb|invariant R| & $\R$ & Temporal refinement mapping \\
    \verb|decreases f| & $f$ & Well-founded metric (progress) \\
  \end{tabular}

  \vspace{0.3cm}
  \verb|done(Q)|: $Q$ holds at termination. \quad
  \verb|now(Q)|: $Q$ holds at some intermediate state.
\end{frame}

\begin{frame}[fragile]{Example: Queue Drain (AF)}
  \begin{block}{Liveness: the queue eventually becomes empty}
  \end{block}
  \vspace{-0.2cm}
  \begin{lstlisting}
fn drain(q: &mut Queue)
    requires q.view().len() > 0,
    ensures  af(done(q.view().len() == 0)),
{
    while q.len() > 0
        invariant q.view().len() >= 0,
        decreases q.view().len(),
    {
        let _ = q.dequeue();
    }
}
  \end{lstlisting}
  \vspace{-0.2cm}
  \verb|invariant| $= \R$, \quad
  \verb|decreases| $= f$ \quad $\Rightarrow$ \quad WhileAU$_L$ rule applies.
\end{frame}

\begin{frame}[fragile]{Example: Bounded Counter (AG)}
  \begin{block}{Safety: the counter stays bounded forever}
  \end{block}
  \vspace{-0.2cm}
  \begin{lstlisting}
fn bounded_counter(x: &mut u64)
    requires *x == 0,
    ensures  ag(*x <= 15),
{
    loop
        invariant *x <= 10,
    {
        if *x < 10 { *x = *x + 1; }
        else       { *x = 0; }
    }
}
  \end{lstlisting}
  \vspace{-0.2cm}
  No \verb|decreases| $\Rightarrow$ infinite loop $\Rightarrow$ WhileAG rule. \\
  \verb|invariant| $(\ast x \leq 10) \implies$ postcondition $(\ast x \leq 15)$.
\end{frame}

\begin{frame}[fragile]{Example: Bind Rule (AG + AF callee)}
  \begin{block}{AG caller invokes AF callee --- bind rule extracts postcondition}
  \end{block}
  \vspace{-0.2cm}
  \begin{lstlisting}
fn reset_to_zero(x: &mut u64)
    ensures af(done(*x == 0)),
{ *x = 0; }

fn ag_caller(x: &mut u64)
    requires *x == 0,
    ensures  ag(*x <= 10),
{
    loop
        invariant *x <= 10,
    {
        if *x < 10 { *x = *x + 1; }
        if *x > 5  { reset_to_zero(x); }
    }
}
  \end{lstlisting}
  \vspace{-0.2cm}
  After \verb|reset_to_zero|: assume $\ast x = 0$ (bind rule $\R$). Invariant preserved.
\end{frame}

\begin{frame}[fragile]{Example: Round-Robin Fairness (AG(AF(now)))}
  \begin{block}{Every element eventually reaches the head of the queue}
  \end{block}
  \vspace{-0.2cm}
  \begin{lstlisting}
fn round_robin(queue: &mut Queue, Ghost(x): Ghost<u64>)
    requires
        queue.view().len() > 1,
        queue.view().contains(x),
    ensures
        ag(queue.view().len() > 0),
        ag(af(now(queue.peek_spec() == x))),
{
    loop
        invariant
            queue.view().len() > 1,
            queue.view().contains(x),
        decreases queue.view().index_of_first(x),
    {
        let val = queue.dequeue();
        queue.enqueue(val);
    }
}
  \end{lstlisting}
  \vspace{-0.3cm}
  {\footnotesize \verb|now()| = state predicate (head $= x$ at loop start, not end). Ghost accumulator tracks it.}
\end{frame}

\begin{frame}[fragile]{Example: Lock Release Liveness (AG(AF(now)))}
  \begin{block}{A held lock is always eventually released}
  \end{block}
  \vspace{-0.2cm}
  \begin{lstlisting}
fn lock_holder(lock: &mut u64, id: u64)
    requires id > 0, *lock == 0,
    ensures  ag(af(now(*lock == 0))),
{
    loop
        invariant *lock == 0 || *lock == id,
        decreases (if *lock == id { 1int } else { 0int }),
    {
        if *lock == 0 {
            *lock = id;    // acquire
            *lock = 0;     // release
        }
        if *lock == id {
            *lock = 0;     // release
        }
    }
}
  \end{lstlisting}
  \vspace{-0.3cm}
  {\footnotesize Single-process liveness. \verb|now(*lock == 0)| holds at release point.}
\end{frame}

\begin{frame}{Complete Rule Set}
  \begin{center}
  \small
  \begin{tabular}{l l}
    \textbf{Construct} & \textbf{Rule} \\
    \hline
    $\rmut{x}{e}$ & State update \\
    $\rlet{x}{t}{u}$ & BindAU$_L$ \\
    $\rif{c}{t}{u}$ & Case split (effectful $c$) \\
    $\rwhile{c}{t}$ + \texttt{decreases} & WhileAU$_L$ \\
    $\rwhile{c}{t}$ (no exit) & WhileAG (coinductive) \\
    $\rcall{f}{\vec{e}}$ & Contract-dependent \\
  \end{tabular}
  \end{center}

  \vspace{0.3cm}
  $\wpx{t}{\varphi}$ for sequential Rust.
  Total correctness Hoare logic is the special case $\AF{\done{Q}}$.
\end{frame}
