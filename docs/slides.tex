\section{Background: TICL over Imp}
% ============================================================================

\begin{frame}{TICL Structural Lemmas for Imp}
  \begin{block}{Core idea (from the TICL paper)}
    For a small imperative language \stimp with mutable state,
    the entailment $\sentailsL{S}{t}{m}{\varphi}$ is proved
    structurally via inference rules on the syntax of $t$.
  \end{block}

  \vspace{0.3cm}
  Key rules from \texttt{imp-lemmas.tex}:
  \begin{itemize}
    \item \textbf{Seq}$_S$AU$_L$: sequential composition decomposes into bind
    \item \textbf{If}$_S$AU$_L$: conditional branches both satisfy the obligation
    \item \textbf{While}$_S$AG: infinite loop with coinductive invariant $\R$
    \item \textbf{While}$_S$AU$_L$: terminating loop with ranking function $f$
  \end{itemize}

  \vspace{0.3cm}
  \alert{Goal:} Lift these to Rust syntax $\to$ define $\wpx{\_}{\_}$ for Vera.
\end{frame}

\begin{frame}{Notation}
  \begin{align*}
    \sigma &\in \text{State (mutable references, heap)} \\
    \ventailsL{t}{\sigma}{\varphi} &\triangleq \text{``Rust term } t \text{ from state } \sigma \text{ satisfies } \varphi\text{''} \\
    \wpx{t}{\varphi} &\triangleq \text{weakest precondition on } \sigma \text{ s.t.\ } \ventailsL{t}{\sigma}{\varphi}
  \end{align*}

  \begin{block}{Temporal formulas}
    \begin{itemize}
      \item $\varphi ::= \now{P} \mid \AG{\varphi} \mid \AU{\varphi}{\varphi'} \mid \varphi \land \varphi'$
      \item Sugar: $\AF{\varphi'} = \AU{\top}{\varphi'}$, \quad $\AX{\varphi'} = \AN{\top}{\varphi'}$
      \item $\done{Q}$: termination postcondition; \quad $\now{P}$: intermediate state predicate
    \end{itemize}
  \end{block}
\end{frame}

\section{Inference Rules}
% ============================================================================

\begin{frame}{Assignment}
  \begin{block}{Rust: \quad $\rmut{x}{e}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=Assign$_\text{Rust}$AU$_L$]
    {\sigma' = \sigma[x \mapsto v] \\
     \varphi(\sigma') \\
     \varphi'(\sigma') \lor \text{(continue checking)}}
    {\ventailsL{\rmut{x}{e}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
  Analogous to \textbf{Assign} in Imp: $\sembra{S}{\IAssign{s}{x}}$ updates the state map.
\end{frame}

\begin{frame}{Sequential Composition (let-binding)}
  \begin{block}{Rust: \quad $\rlet{x}{t}{u}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=Let$_\text{Rust}$AU$_L$]
    {\ventailsR{t}{\sigma}{\AU{\varphi}{\AX{\done{\R}}}} \\\\
     \forall~x,\sigma',~\R~x~\sigma' \to \ventailsL{u[x]}{\sigma'}{\AU{\varphi}{\varphi'}}}
    {\ventailsL{\rlet{x}{t}{u}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
  Direct lift of \textbf{BindAU$_L$} / \textbf{Seq}$_S$AU$_L$ from TICL.
\end{frame}

\begin{frame}{Conditional}
  \begin{block}{Rust: \quad $\rif{c}{t}{u}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=If$_\text{Rust}$AU$_L$]
    {\ventailsR{c}{\sigma}{\AU{\varphi}{\doneq{(b,\sigma')}{}}} \\\\
     {
       \begin{cases}
         \ventailsL{t}{\sigma'}{\AU{\varphi}{\varphi'}}, &\text{if } b \\
         \ventailsL{u}{\sigma'}{\AU{\varphi}{\varphi'}}, &\text{otherwise}
       \end{cases}
     }
    }
    {\ventailsL{\rif{c}{t}{u}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
  Evaluating $c$ may take multiple steps; $\varphi$ must hold along the path.
\end{frame}

\begin{frame}{While Loop (AU -- terminating)}
  \begin{block}{Rust: \quad $\rwhile{c}{t}$ \quad with \texttt{invariant} $\R$, \texttt{decreases} $f$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=While$_\text{Rust}$AU$_L$]
    {\R~\sigma \\\\
     {\begin{aligned}
      \forall~\sigma,~&\R~\sigma \to \ventailsR{c}{\sigma}{\AU{\varphi}{\doneq{(b,\sigma')}{}}} \\
      & \land \begin{cases}
        \ventailsL{t}{\sigma'}{\AU{\varphi}{\varphi'}} ~\lor \\
        \quad \ventailsR{t}{\sigma'}{\AU{\varphi}{\AF{\done{(\lambda~\sigma'' .~
            \R~\sigma'' \land f~\sigma'' < f~\sigma)}}}}, &\text{if } b \\
        \varphi'(\sigma'), &\text{otherwise}
      \end{cases}
     \end{aligned}}}
    {\ventailsL{\rwhile{c}{t}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
\end{frame}

\begin{frame}{Loop (AG -- infinite)}
  \begin{block}{Rust: \quad $\rloop{t}$ \quad with \texttt{invariant} $\R$ (no \texttt{decreases})}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=Loop$_\text{Rust}$AG]
    {\R~\sigma \\\\
     {\begin{aligned}
      \forall~\sigma,~&\R~\sigma \to \\
      & \ventailsL{\rloop{t}}{\sigma}{\varphi} ~\land \\
      & \ventailsR{t}{\sigma}{\varphi~\AU{}~\AF{\done{(\lambda~\sigma' \Rightarrow \R~\sigma')}}}
     \end{aligned}}}
    {\ventailsL{\rloop{t}}{\sigma}{\AG{\varphi}}}
  \end{mathpar}
  }
  Coinductive hypothesis $\ventailsL{\rloop{t}}{\sigma}{\varphi}$ enables proving $\varphi$
  holds at loop re-entry after each iteration.
\end{frame}

\begin{frame}{Async Bind (Lazy Future Creation)}
  \begin{block}{Rust: \quad $\rlet{p}{\rasync{\rcall{f}{\vec{e}}}}{k~p}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=AsyncBind$_\text{Rust}$]
    {\text{No state change (lazy future)} \\\\
     \ventailsL{k~p}{\sigma}{\Phi}}
    {\ventailsL{\rlet{p}{\rasync{\rcall{f}{\vec{e}}}}{k~p}}{\sigma}{\Phi}}
  \end{mathpar}
  }
  Rust futures are lazy: \texttt{async fn()} returns a handle, body runs at \texttt{.await}.
\end{frame}

\begin{frame}{Await (AU/AF -- terminating callee)}
  \begin{block}{Rust: \quad $\rlet{x}{\rawait{p}}{k~x}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=Await$_\text{Rust}$AU$_L$]
    {\ventailsR{P[p]}{\sigma}{\AU{\varphi'}{\AX{\done{\R}}}} \\\\
     \varphi' \implies \varphi \quad\text{(transparent)} \\\\
     \forall~x,\sigma',~\R~x~\sigma' \to \ventailsL{k~x}{\sigma'}{\AU{\varphi}{\varphi''}}}
    {\ventailsL{\rlet{x}{\rawait{p}}{k~x}}{\sigma}{\AU{\varphi}{\varphi''}}}
  \end{mathpar}
  }
  $P[p]$: async process behind future $p$.
  Callee's path $\varphi'$ must imply caller's $\varphi$ (transparent await).
\end{frame}

\begin{frame}{Await (AG -- diverging callee)}
  \begin{block}{Rust: \quad $\rawait{p}$ \quad where $P[p]$ satisfies $\AG{\psi}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=Await$_\text{Rust}$AG]
    {\ventailsL{P[p]}{\sigma}{\AG{\psi}} \\\\
     \psi \implies \varphi}
    {\ventailsL{\rawait{p}}{\sigma}{\AG{\varphi}}}
  \end{mathpar}
  }
  Callee runs forever $\to$ await is a \emph{divergence point}.
  Caller's $\AG{\varphi}$ discharged if $\psi \implies \varphi$.
\end{frame}

% ============================================================================
\section{Weakest Preconditions}
% ============================================================================

\begin{frame}{wp: Assignment}
    $$\wpx{\rmut{x}{e}}{\AU{\varphi}{\varphi'}} ~=~ \varphi[\sigma'/\sigma] \land (\varphi'[\sigma'/\sigma] \lor \top)$$
    where $\sigma' = \sigma[x \mapsto \llbracket e \rrbracket_\sigma]$.
\end{frame}

\begin{frame}{wp: Let-binding (Sequential Composition)}
    $$\wpx{\rlet{x}{t}{u}}{\AU{\varphi}{\varphi'}} ~=~ \wpx{t}{\AU{\varphi}{\AX{\done{\R}}}}$$
    $$\land~ \forall x, \sigma'.~ \R~x~\sigma' \to \wpx{u[x]}{\AU{\varphi}{\varphi'}}$$
\end{frame}

\begin{frame}{wp: Conditional}
    \begin{align*}
    \wpx{\rif{c}{t}{u}}{\AU{\varphi}{\varphi'}} ~=~
    &~ \wpx{c}{\AU{\varphi}{\doneq{(b,\sigma')}{}}} \\
    \land~& \begin{cases}
      \wpx{t}{\AU{\varphi}{\varphi'}}[\sigma'/\sigma], &\text{if } b \\
      \wpx{u}{\AU{\varphi}{\varphi'}}[\sigma'/\sigma], &\text{otherwise}
    \end{cases}
    \end{align*}
\end{frame}

\begin{frame}{wp: While Loop (AU)}
    \begin{align*}
    \wpx{\rwhile{c}{t}}{\AU{\varphi}{\varphi'}} ~=~
    & ~\R~\sigma ~\land~ \forall~\sigma,~\R~\sigma \to \\
    & ~\wpx{c}{\AU{\varphi}{\doneq{(b,\sigma')}{}}} ~\land \\
    & \begin{cases}
      \wpx{t}{\AU{\varphi}{\varphi'}}[\sigma'/\sigma] ~\lor \\
      \quad \wpx{t}{\AU{\varphi}{\AF{\done{(\lambda~\sigma''.~
          \R~\sigma'' \land f~\sigma'' < f~\sigma)}}}}& \\
      \quad\quad [\sigma'/\sigma], &\text{if } b \\
      \varphi'(\sigma'), &\text{o.w.}
    \end{cases}
    \end{align*}
\end{frame}

\begin{frame}{wp: Loop (AG)}
    \begin{align*}
    \wpx{\rloop{t}}{\AG{\varphi}} ~=~
    & ~\R~\sigma ~\land~ \forall~\sigma,~\R~\sigma \to \\
    & ~\ventailsL{\rloop{t}}{\sigma}{\varphi} ~\land \\
    & ~\wpx{t}{\varphi~\AU{}~\AF{\done{(\lambda~\sigma'.~ \R~\sigma')}}}
    \end{align*}
    Coinductive: $\ventailsL{\rloop{t}}{\sigma}{\varphi}$ is the coinductive hypothesis.
\end{frame}

\begin{frame}{wp: Function Call -- $f$ ensures $\AF{\done{\R_f}}$}
  \begin{block}{$f$ has terminating postcondition}
    $$\wpx{\rcall{f}{e_1, \ldots, e_i}}{\AU{\varphi}{\varphi'}} ~=~ \forall~x,\sigma'.~ \R_f~x~\sigma' \to \wpx{k~x}{\AU{\varphi}{\varphi'}}$$
    where $k$ is the continuation after the call.
  \end{block}

  \begin{block}{$f$ has temporal ensures $\Psi$: transparent checking}
    Additionally assert $\Psi \implies \Phi$ where $\Phi$ is the caller's obligation.
  \end{block}

  \begin{block}{$f$ ensures $\AG{\psi}$: cannot prove $\AF{}$ for it}
    If $f$ has $\AG{\psi}$ (diverges), then $\wpx{\rcall{f}{\vec{e}}}{\AF{\varphi'}}$ is \textbf{unprovable} --
    a diverging call can never reach the $\AF{}$ goal.
  \end{block}
\end{frame}

\begin{frame}{wp: Async Bind}
    $$\wpx{\rlet{p}{\rasync{\rcall{f}{\vec{e}}}}{k~p}}{\Phi} ~=~ \wpx{k~p}{\Phi}$$
    No-op: lazy future creation does not change state.
\end{frame}

\begin{frame}{wp: Await (AU)}
    \begin{align*}
    \wpx{\rlet{x}{\rawait{p}}{k~x}}{\AU{\varphi}{\varphi''}} ~=~
    & ~\varphi' \implies \varphi \\
    \land~& ~\forall~x,\sigma'.~ \R~x~\sigma' \to \wpx{k~x}{\AU{\varphi}{\varphi''}}
    \end{align*}
    where $P[p]~\rust{ensures}~\AU{\varphi'}{\AX{\done{\R}}}$.
\end{frame}

\begin{frame}{wp: Await (AG)}
    $$\wpx{\rawait{p}}{\AG{\varphi}} ~=~ \psi \implies \varphi$$
    where $P[p]~\rust{ensures}~\AG{\psi}$.

    \vspace{0.3cm}
    Callee diverges $\to$ continuation is dead code $\to$ no further wp.
\end{frame}

\section{Summary}
% ============================================================================

\begin{frame}{Complete Rule Set}
  \begin{center}
  \small
  \begin{tabular}{l l l}
    \textbf{Rust Construct} & \textbf{Rule} & \textbf{Section} \\
    \hline
    $\rmut{x}{e}$ & State update, check $\varphi$ & Inference \\
    $\rlet{x}{t}{u}$ & BindAU$_L$ & Inference \\
    $\rif{c}{t}{u}$ & Case split (effectful $c$) & Inference \\
    $\rwhile{c}{t}$ + \texttt{decreases} & IterAU$_L$ & Inference \\
    $\rloop{t}$ (no exit) & IterAG (coinductive) & Inference \\
    $\rlet{p}{\rasync{\rcall{f}{\vec{e}}}}{k}$ & No-op (lazy) & Inference \\
    $\rawait{p}$ (AU callee) & Bind + transparent & Inference \\
    $\rawait{p}$ (AG callee) & Divergence & Inference \\
    \hline
    $\rcall{f}{\vec{e}}$ (AF ensures) & Bind + assume $\R_f$ & WP only \\
    $\rcall{f}{\vec{e}}$ (AG ensures) & Transparent $\Psi \implies \Phi$ & WP only \\
    $\rcall{f}{\vec{e}}$ (AG + AF goal) & \textbf{Unprovable} & WP only \\
  \end{tabular}
  \end{center}

  \vspace{0.3cm}
  Function calls are contract-dependent $\to$ only in wp, not inference rules.
\end{frame}
