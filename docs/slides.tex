\section{Background: TICL over Imp}
% ============================================================================

\begin{frame}{TICL Structural Lemmas for Imp}
  \begin{block}{Core idea (from the TICL paper)}
    For a small imperative language \stimp with mutable state,
    the entailment $\sentailsL{S}{t}{m}{\varphi}$ is proved
    structurally via inference rules on the syntax of $t$.
  \end{block}

  \vspace{0.3cm}
  Key rules from \texttt{imp-lemmas.tex}:
  \begin{itemize}
    \item \textbf{Seq}$_S$AU$_L$: sequential composition decomposes into bind
    \item \textbf{If}$_S$AU$_L$: conditional branches both satisfy the obligation
    \item \textbf{While}$_S$AG: infinite loop with coinductive invariant $\R$
    \item \textbf{While}$_S$AU$_L$: terminating loop with ranking function $f$
  \end{itemize}

  \vspace{0.3cm}
  \alert{Goal:} Lift these to Rust syntax $\to$ define $\wpx{\_}{\_}$ for Vera.
\end{frame}

\begin{frame}{Notation}
  \begin{align*}
    \sigma &\in \text{State (mutable references, heap)} \\
    \ventailsL{t}{\sigma}{\varphi} &\triangleq \text{``Rust term } t \text{ from state } \sigma \text{ satisfies } \varphi\text{''} \\
    \wpx{t}{\varphi} &\triangleq \text{weakest precondition on } \sigma \text{ s.t.\ } \ventailsL{t}{\sigma}{\varphi}
  \end{align*}

  \begin{block}{Temporal formulas}
    \begin{itemize}
      \item $\varphi ::= \now{P} \mid \AG{\varphi} \mid \AU{\varphi}{\varphi'} \mid \varphi \land \varphi'$
      \item Sugar: $\AF{\varphi'} = \AU{\top}{\varphi'}$, \quad $\AX{\varphi'} = \AN{\top}{\varphi'}$
      \item $\done{Q}$: termination postcondition; \quad $\now{P}$: intermediate state predicate
    \end{itemize}
  \end{block}
\end{frame}

\section{Inference Rules (Single-Process)}
% ============================================================================

\begin{frame}{Assignment}
  \begin{block}{Rust: \quad $\rmut{x}{e}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=Assign$_\text{Rust}$AU$_L$]
    {\ventailsR{e}{\sigma}{\AU{\varphi}{\done{\R}}} \\\\
     \forall~v, \sigma',~\R~v~\sigma' \to \ventailsL{\ISkip}{\sigma'[x \mapsto v]}{\AU{\varphi}{\varphi'}}}
    {\ventailsL{\rmut{x}{e}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
  Evaluating $e$ may take steps; $\varphi$ must hold along the path.
\end{frame}

\begin{frame}{Sequential Composition (let-binding)}
  \begin{block}{Rust: \quad $\rlet{x}{t}{u}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=Let$_\text{Rust}$AU$_L$]
    {\ventailsR{t}{\sigma}{\AU{\varphi}{\done{\R}}} \\\\
     \forall~x,\sigma',~\R~x~\sigma' \to \ventailsL{u[x]}{\sigma'}{\AU{\varphi}{\varphi'}}}
    {\ventailsL{\rlet{x}{t}{u}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
  Direct lift of \textbf{BindAU$_L$} / \textbf{Seq}$_S$AU$_L$ from TICL.
\end{frame}

\begin{frame}{Conditional}
  \begin{block}{Rust: \quad $\rif{c}{t}{u}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=If$_\text{Rust}$AU$_L$]
    {\ventailsR{c}{\sigma}{\AU{\varphi}{\doneq{(b,\sigma')}{}}} \\\\
     {
       \begin{cases}
         \ventailsL{t}{\sigma'}{\AU{\varphi}{\varphi'}}, &\text{if } b \\
         \ventailsL{u}{\sigma'}{\AU{\varphi}{\varphi'}}, &\text{otherwise}
       \end{cases}
     }
    }
    {\ventailsL{\rif{c}{t}{u}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
  Evaluating $c$ may take multiple steps; $\varphi$ must hold along the path.
\end{frame}

\begin{frame}{While Loop (AU -- terminating)}
  \begin{block}{Rust: \quad $\rwhile{c}{t}$ \quad with \texttt{invariant} $\R$, \texttt{decreases} $f$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=While$_\text{Rust}$AU$_L$]
    {\R~\sigma \\\\
     {\begin{aligned}
      \forall~\sigma,~&\R~\sigma \to \ventailsR{c}{\sigma}{\AU{\varphi}{\doneq{(b,\sigma')}{}}} \\
      & \land \begin{cases}
        \ventailsL{t}{\sigma'}{\AU{\varphi}{\varphi'}} ~\lor \\
        \quad \ventailsR{t}{\sigma'}{\AU{\varphi}{\done{(\lambda~\sigma'' .~
            \R~\sigma'' \land f~\sigma'' < f~\sigma)}}}, &\text{if } b \\
        \varphi'(\sigma'), &\text{otherwise}
      \end{cases}
     \end{aligned}}}
    {\ventailsL{\rwhile{c}{t}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
\end{frame}

\begin{frame}{While Loop (AG -- infinite)}
  \begin{block}{Rust: \quad $\rwhile{c}{t}$ \quad with \texttt{invariant} $\R$ (no \texttt{decreases})}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=While$_\text{Rust}$AG]
    {\R~\sigma \\\\
     {\begin{aligned}
      \forall~\sigma,~&\R~\sigma \to \\
      & \ventailsL{\rwhile{c}{t}}{\sigma}{\varphi} ~\land \\
      & \ventailsR{c}{\sigma}{\AU{\varphi}{\doneq{(\tm{true},\sigma)}{}}} ~\land \\
      & \ventailsR{t}{\sigma}{\varphi~\AU{}~\done{(\lambda~\sigma' \Rightarrow \R~\sigma')}}
     \end{aligned}}}
    {\ventailsL{\rwhile{c}{t}}{\sigma}{\AG{\varphi}}}
  \end{mathpar}
  }
  Coinductive hypothesis $\ventailsL{\rwhile{c}{t}}{\sigma}{\varphi}$; condition must always be true.
\end{frame}

\begin{frame}{Function Call (Bind Rule)}
  \begin{block}{Rust: \quad $\rlet{x}{\rcall{f}{\vec{e}}}{k~x}$}
  \end{block}
  {\footnotesize
  \begin{mathpar}
    \inferrule*[Right=Call$_\text{Rust}$AU$_L$]
    {\ventailsR{\rcall{f}{\vec{e}}}{\sigma}{\AU{\varphi}{\done{\R}}} \\\\
     \forall~x,\sigma',~\R~x~\sigma' \to \ventailsL{k~x}{\sigma'}{\AU{\varphi}{\varphi'}}}
    {\ventailsL{\rlet{x}{\rcall{f}{\vec{e}}}{k~x}}{\sigma}{\AU{\varphi}{\varphi'}}}
  \end{mathpar}
  }
  Instance of Let rule. Call is a sub-expression that produces a value.
\end{frame}

\section{Single-Process Weakest Preconditions}
% ============================================================================

\begin{frame}{wp: Assignment}
    \begin{align*}
    \wpx{\rmut{x}{e}}{\AU{\varphi}{\varphi'}} ~=~ \lambda~\sigma.~
    &~ \wpx{e}{\AU{\varphi}{\done{\R}}}~\sigma \\
    \land~& ~\forall~v, \sigma'.~ \R~v~\sigma' \to \varphi'(\sigma'[x \mapsto v])
    \end{align*}
\end{frame}

\begin{frame}{wp: Let-binding (Sequential Composition)}
    \begin{align*}
    \wpx{\rlet{x}{t}{u}}{\AU{\varphi}{\varphi'}} ~=~ \lambda~\sigma.~
    &~ \wpx{t}{\AU{\varphi}{\done{\R}}}~\sigma \\
    \land~& ~\forall x, \sigma'.~ \R~x~\sigma' \to \wpx{u[x]}{\AU{\varphi}{\varphi'}}~\sigma'
    \end{align*}
\end{frame}

\begin{frame}{wp: Conditional}
    \begin{align*}
    &\wpx{\rif{c}{t}{u}}{\AU{\varphi}{\varphi'}} ~=~ \lambda~\sigma. \\
    & \quad \wpx{c}{\AU{\varphi}{\done{\R}}}~\sigma ~\land~ \forall~b,\sigma'.~ \R~b~\sigma' \to \\
    & \quad \begin{cases}
      \wpx{t}{\AU{\varphi}{\varphi'}}~\sigma', &\text{if } b \\
      \wpx{u}{\AU{\varphi}{\varphi'}}~\sigma', &\text{otherwise}
    \end{cases}
    \end{align*}
\end{frame}

\begin{frame}{wp: While Loop (AU)}
    \begin{align*}
    &\wpx{\rwhile{c}{t}}{\AU{\varphi}{\varphi'}} ~=~ \lambda~\sigma. \\
    & \quad \R~\sigma ~\land~ \forall~\sigma,~\R~\sigma \to \\
    & \quad \wpx{c}{\AU{\varphi}{\done{\R'}}}~\sigma ~\land~ \forall~b,\sigma'.~ \R'~b~\sigma' \to \\
    & \quad \begin{cases}
      \wpx{t}{\AU{\varphi}{\varphi'}}~\sigma' ~\lor \\
      \quad \wpx{t}{\AU{\varphi}{\done{(\lambda~\_~\sigma''.~
          \R~\sigma'' \land f~\sigma'' < f~\sigma)}}}~\sigma', &\text{if } b \\
      \varphi'(\sigma'), &\text{o.w.}
    \end{cases}
    \end{align*}
\end{frame}

\begin{frame}{wp: While Loop (AG)}
    \begin{align*}
    &\wpx{\rwhile{c}{t}}{\AG{\varphi}} ~=~ \lambda~\sigma. \\
    & \quad \R~\sigma ~\land~ \forall~\sigma,~\R~\sigma \to \\
    & \quad \wpx{\rwhile{c}{t}}{\varphi}~\sigma ~\land \\
    & \quad \wpx{c}{\AU{\varphi}{\doneq{(\tm{true},\sigma)}{}}}~\sigma ~\land \\
    & \quad \wpx{t}{\varphi~\AU{}~\done{(\lambda~\sigma'.~ \R~\sigma')}}~\sigma
    \end{align*}
    Coinductive: $\wpx{\rwhile{c}{t}}{\varphi}~\sigma$ is the coinductive hypothesis.
\end{frame}

\begin{frame}{wp: Function Call -- $f$ ensures $\AF{\done{\R_f}}$}
  \begin{block}{$f$ has terminating postcondition}
    $$\wpx{\rcall{f}{e_1, \ldots, e_i}}{\AU{\varphi}{\varphi'}} ~=~ \lambda~\sigma.~ \forall~x,\sigma'.~ \R_f~x~\sigma' \to \wpx{k~x}{\AU{\varphi}{\varphi'}}~\sigma'$$
    where $k$ is the continuation after the call.
  \end{block}

  \begin{block}{$f$ has temporal ensures $\Psi$: transparent checking}
    Additionally assert $\Psi \implies \Phi$ where $\Phi$ is the caller's obligation.
  \end{block}

  \begin{block}{$f$ ensures $\AG{\psi}$: cannot prove $\AF{}$ for it}
    If $f$ has $\AG{\psi}$ (diverges), then $\wpx{\rcall{f}{\vec{e}}}{\AF{\varphi'}}$ is \textbf{unprovable} --
    a diverging call can never reach the $\AF{}$ goal.
  \end{block}
\end{frame}

\section{Multi-Process Configuration}
% ============================================================================

\begin{frame}{Configuration}
  A \textbf{configuration} $C = \Config{P}{\sigma}{i}$ consists of:

  \begin{align*}
    P &: \PID \to \text{Term} && \text{process dictionary} \\
    \sigma &\in \text{State} && \text{shared mutable state (heap, \&mut refs)} \\
    i &\in \PID && \text{currently active (scheduled) process}
  \end{align*}

  \begin{block}{Cooperative scheduling}
    \begin{itemize}
      \item Each process runs until it hits \texttt{.await}, then yields
      \item The scheduler picks the next process from the ready set
      \item Temporal formulas $\varphi$ are over shared state $\sigma$
    \end{itemize}
  \end{block}

  \begin{block}{Multi-process wp}
    $\WPx{C}{\varphi}$ -- weakest precondition on configuration $C$.
    Defined by cases on the shape of $P(i)$.
  \end{block}
\end{frame}

\section{Multi-Process Weakest Preconditions}
% ============================================================================

\begin{frame}{WP: Sequential Step}
  When $P(i)$ is a non-async construct (assignment, let, if, while, sync call):

  $$\WPx{\Config{P}{\sigma}{i}}{\varphi} ~=~ \wpx{P(i)}{\varphi}~\sigma$$

  Delegate to single-process $\wpx{}{}$. Process dictionary $P$ is unchanged.

  \vspace{0.5cm}
  Programs without \texttt{async}/\texttt{await} reduce entirely to single-process wp:
  $$\WPx{\Config{\{0 \mapsto t\}}{\sigma}{0}}{\varphi} ~=~ \wpx{t}{\varphi}~\sigma$$
\end{frame}

\begin{frame}{WP: Async Bind (Lazy Future Creation)}
  \begin{block}{Rust: \quad $P(i) = \rlet{p}{\rasync{\rcall{f}{\vec{e}}}}{k~p}$}
  \end{block}

  $$\WPx{\Config{P}{\sigma}{i}}{\Phi} ~=~ \WPx{\Config{P[i \mapsto k,~p \mapsto f_{\mathit{body}}]}{\sigma}{i}}{\Phi}$$

  \begin{itemize}
    \item Creates new process $p$ with body $f_{\mathit{body}}$ (suspended)
    \item Replaces active process $i$'s term with continuation $k$
    \item No state change, no scheduling -- futures are lazy
  \end{itemize}
\end{frame}

\begin{frame}{WP: Await (AU -- terminating callee)}
  \begin{block}{Rust: \quad $P(i) = \rlet{x}{\rawait{p}}{k~x}$}
  \end{block}

  \begin{align*}
  &\WPx{\Config{P}{\sigma}{i}}{\AU{\varphi}{\varphi'}} ~= \\
  & \quad \WPx{\Config{P}{\sigma}{p}}{\AU{\varphi}{\done{\R}}} \\
  & \quad \land~ \forall~x, \sigma'.~ \R~x~\sigma' \to \WPx{\Config{P[i \mapsto k[x]]}{\sigma'}{i}}{\AU{\varphi}{\varphi'}}
  \end{align*}

  \begin{itemize}
    \item Switch active process to $p$; run it maintaining $\varphi$
    \item After $p$ terminates with result $x$ and state $\sigma'$, resume $i$
  \end{itemize}
\end{frame}

\begin{frame}{WP: Await (AG -- diverging callee)}
  \begin{block}{Rust: \quad $P(i) = \rawait{p}$ \quad where $P(p)$ satisfies $\AG{\psi}$}
  \end{block}

  $$\WPx{\Config{P}{\sigma}{i}}{\AG{\varphi}} ~=~ \WPx{\Config{P}{\sigma}{p}}{\AG{\psi}} ~\land~ (\psi \implies \varphi)$$

  \begin{itemize}
    \item Process $p$ runs forever (AG); await diverges
    \item Caller's $\AG{\varphi}$ discharged if $\psi \implies \varphi$
    \item Continuation after await is unreachable
  \end{itemize}
\end{frame}

\section{Summary}

\begin{frame}{Complete Rule Set}
  \begin{center}
  \small
  \begin{tabular}{l l l}
    \textbf{Construct} & \textbf{Rule} & \textbf{Layer} \\
    \hline
    $\rmut{x}{e}$ & State update & $\wpx{}{}$ \\
    $\rlet{x}{t}{u}$ & BindAU$_L$ & $\wpx{}{}$ \\
    $\rif{c}{t}{u}$ & Case split (effectful $c$) & $\wpx{}{}$ \\
    $\rwhile{c}{t}$ + \texttt{dec.} & While$_\text{Rust}$AU$_L$ & $\wpx{}{}$ \\
    $\rwhile{c}{t}$ (no exit) & While$_\text{Rust}$AG & $\wpx{}{}$ \\
    $\rcall{f}{\vec{e}}$ & Contract-dependent & $\wpx{}{}$ \\
    \hline
    Sequential step & Delegate to $\wpx{}{}$ & $\WPx{}{}$ \\
    $\rust{async}~\rcall{f}{\vec{e}}$ & Extend $P$, no-op & $\WPx{}{}$ \\
    $\rawait{p}$ (AU callee) & Switch active to $p$ & $\WPx{}{}$ \\
    $\rawait{p}$ (AG callee) & Divergence & $\WPx{}{}$ \\
  \end{tabular}
  \end{center}

  \vspace{0.3cm}
  Two layers: $\wpx{t}{\varphi}$ for sequential Rust,
  $\WPx{\Config{P}{\sigma}{i}}{\varphi}$ for async concurrency.
\end{frame}
